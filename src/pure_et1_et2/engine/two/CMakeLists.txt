cmake_minimum_required(VERSION 3.16)
project(IPS_TG_v25 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP)
set(SOURCES main.cpp args.hpp array.hpp ips.hpp graph.hpp hash.hpp heap.hpp io.hpp)
# Four independent binaries, same graph reader and compiler flags.
# Classic pivot=TG OFF; partition-oriented IPS pivot=TG ON.
foreach(variant hybrid_classic compact_classic hybrid_tg compact_tg)
  add_executable(${variant} ${SOURCES})
  target_compile_options(${variant} PRIVATE -O3 -DNDEBUG -march=native -mavx)
  target_compile_definitions(${variant} PRIVATE IPS_ST_V15_HASHLESS IPS_ST_V15_ROW_CACHE)
  if(variant MATCHES "^compact_")
    target_compile_definitions(${variant} PRIVATE IPS_ST_V24_SINGLE_DEFECT)
  endif()
  if(variant MATCHES "_classic$")
    target_compile_definitions(${variant} PRIVATE IPS_CLASSIC_PIVOT)
  endif()
  if(OpenMP_CXX_FOUND)
    target_link_libraries(${variant} PRIVATE OpenMP::OpenMP_CXX)
  endif()
endforeach()
