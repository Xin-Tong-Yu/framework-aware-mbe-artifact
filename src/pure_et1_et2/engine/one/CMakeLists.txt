cmake_minimum_required(VERSION 3.16)
project(mbe_one_splitpool_v32 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(mbet_fast main.cpp)
target_include_directories(mbet_fast PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(mbet_fast PRIVATE /O2 /W3)
else()
  target_compile_options(mbet_fast PRIVATE -O3 -DNDEBUG -Wall -Wextra -Wno-unused-parameter)
endif()
option(MBE_ET_PAIR_GATE "Precheck first two candidate paths" OFF)
option(MBE_ET_STREAM "Streaming terminal verification" OFF)
if(MBE_ET_PAIR_GATE)
  target_compile_definitions(mbet_fast PRIVATE MBE_ET_PAIR_GATE=1)
endif()
if(MBE_ET_STREAM)
  target_compile_definitions(mbet_fast PRIVATE MBE_ET_STREAM=1)
endif()

option(MBE_V29_REUSE_STACK "Reuse fork path stacks by recursion depth" OFF)
option(MBE_V29_REUSE_PREV "Reuse fork previous-node work vectors by recursion depth" OFF)
option(MBE_V29_RESULT_HINT "Give ordered map end hints for result-trie insertions" OFF)
option(MBE_V29_PROFILE "Enable detailed phase timing and counters (not for benchmark)" OFF)
foreach(opt MBE_V29_REUSE_STACK MBE_V29_REUSE_PREV MBE_V29_RESULT_HINT MBE_V29_PROFILE)
  if(${opt})
    target_compile_definitions(mbet_fast PRIVATE ${opt}=1)
  endif()
endforeach()
if(MBE_V29_PROFILE)
  target_compile_definitions(mbet_fast PRIVATE DEBUG=1)
endif()

option(MBE_V30_SKIP_EMPTY_PREV "Skip unconditional cursor resets for paths without predecessors" OFF)
option(MBE_V30_LAZY_CURSOR "Lazy per-prefix fork cursor setup" OFF)
option(MBE_V30_NO_SHRINK "Avoid redundant shrink_to_fit just before vector destruction" OFF)
foreach(opt MBE_V30_SKIP_EMPTY_PREV MBE_V30_LAZY_CURSOR MBE_V30_NO_SHRINK)
  if(${opt})
    target_compile_definitions(mbet_fast PRIVATE ${opt}=1)
  endif()
endforeach()

option(MBE_V31_FORK_PATH "Use the active Find DFS path rather than rebuilding the fork ancestor path" OFF)
option(MBE_V31_RESULT_HOT "Parent-local last child pointer cache for Result Tree" OFF)
option(MBE_V32_CANDIDATE_POOL "Recyclable pool for Candidate NeighborTreeNode objects only" OFF)
option(MBE_V32_RESULT_POOL "Recyclable pool for ResultTreeNode objects only" OFF)
foreach(opt MBE_V31_FORK_PATH MBE_V31_RESULT_HOT MBE_V32_CANDIDATE_POOL MBE_V32_RESULT_POOL)
  if(${opt})
    target_compile_definitions(mbet_fast PRIVATE ${opt}=1)
  endif()
endforeach()

option(MBE_ET13_DIAG "Separate instrumented ET diagnostic binary" OFF)
if(MBE_ET13_DIAG)
  target_compile_definitions(mbet_fast PRIVATE MBE_ET13_DIAG=1)
endif()

# V40 contiguous ordered map A/B switches (off = exactly original V39).
option(MBE_V40_FLAT_RESULT "Use sorted contiguous storage for result trie children" OFF)
option(MBE_V40_FLAT_CANDIDATE "Use sorted contiguous storage for candidate trie children" OFF)
if(MBE_V40_FLAT_RESULT)
  target_compile_definitions(mbet_fast PRIVATE MBE_V40_FLAT_RESULT=1)
endif()
if(MBE_V40_FLAT_CANDIDATE)
  target_compile_definitions(mbet_fast PRIVATE MBE_V40_FLAT_CANDIDATE=1)
endif()

# V41 standalone result-node child-allocation and lookup A/B settings.
option(MBE_V41_POOLED_RESULT_CHILDREN "Use a recyclable allocator for result children arrays" OFF)
set(MBE_V41_LINEAR_RESULT_LIMIT "0" CACHE STRING "Result children linear-scan limit, 0=V40 binary-search")
if(MBE_V41_POOLED_RESULT_CHILDREN)
  target_compile_definitions(mbet_fast PRIVATE MBE_V41_POOLED_RESULT_CHILDREN=1)
endif()
if(NOT MBE_V41_LINEAR_RESULT_LIMIT STREQUAL "0")
  target_compile_definitions(mbet_fast PRIVATE MBE_V41_LINEAR_RESULT_LIMIT=${MBE_V41_LINEAR_RESULT_LIMIT})
endif()
